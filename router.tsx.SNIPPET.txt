Open your existing src/router.tsx and add ONE line to createRouter(...):

  basepath: import.meta.env.BASE_URL.replace(/\/$/, '') || '/',

Example (your file may differ slightly):

  export function getRouter() {
    return createRouter({
      routeTree,
      basepath: import.meta.env.BASE_URL.replace(/\/$/, '') || '/',
      scrollRestoration: true,
    })
  }
